# GLM-5.3 on two Sparks: copy to .env.glm (read by scripts/glm-up.sh; scripts/dual-down.sh stops it). docs/glm53-plan.md.
# Checkpoint at the same path on both nodes.
GLM_MODEL_DIR=/home/ryan/models/GLM-5.3-EXL3-3.0bpw
PORT=8000
SERVED_MODEL_NAME=glm-5.3
# Context. The MLA latent cache in FP8, split across the ranks (GLM_KV_SPLIT below): 256K is
# 7.3 GB a rank. The FP4/FP8 mix (GLM_KV=fp4, GLM_KV_FP8_LAYERS=0-9,68-77) fit 320K in less, at
# KL +0.004 on the long eval, but in use the model's recall of its context was visibly worse
# (2026-10-20), so the profile stays FP8; the mix remains available as a setting.
GLM_MAX_SEQ=262144
# 4,096-row chunks with the EXL3 prefill kernel that decodes each expert's trellis once a 64-row
# block (GLM_EXL3_PREFILL below): 8K prompt 18.9 -> 17.8 s, the MoE 10.7 -> 9.8 s, for ~1 GB more
# transient memory during a prefill (2026-10-20, docs/gotchas.md). 2,048 holds that GB back.
GLM_CHUNK=4096
# Expert slots per rank (auto: MemAvailable less a 20 GB floor, 88% of the rest).
# 8,600 routed slots (44%): Tokyo 3-row eval KL 0.171 at 7,000 -> 0.132 at 8,500 -> 0.113 at 9,394.
# Measured live 2026-10-16 with scripts/mem-watchdog.sh: min MemAvailable 6.0 GB (rank0, an
# hf download running) / 8.2 GB (rank1, polkitd + bluetoothd leaking 6.4 GB) through 20K and 28K
# prompts. Raise toward ~9,200 once both are gone; auto (20 GB floor) gave ~7,100.
# FP8 KV cache: +0.002-0.005 NLL over BF16 and half its memory (~900 EXL3 slots at 128K).
GLM_KV=fp8
# Each rank caches every other position (docs/gotchas.md): half the KV memory, decode at parity.
GLM_KV_SPLIT=1
# A prefill chunk's attention partials summed over the ranks in BF16 instead of fp32: 8K prompt
# 0.75 s less attention time (18.9 -> 18.2 s), one BF16 step on each layer's attention output;
# against the fp32 sum, every expert exact, long set: KL 0.014 / top-1 97.3%, the same as a
# one-bit control (docs/gotchas.md 2026-10-20). rs: the fp32 sum at three quarters of the bytes.
GLM_PREFILL_REDUCE=bf16
# Prompt prefixes on disk: another chat or Open WebUI's title/tag requests no longer cost a long
# conversation's next turn a full prefill. Rank 0's drive had 62 GB free (2026-10-17).
GLM_PREFIX_DISK=1
GLM_PREFIX_DISK_GB=20
# 12,000 while rank 1's polkitd/bluetoothd leak holds ~3.9 GB (2026-10-18: 12,350 took rank 1 to
# 2.94 GiB under the watchdog's floor). Back to 12,350 after `systemctl restart polkit` there.
# 12,400 since 2026-10-20, after rank 1's polkitd (4.7 GB, fed by a dashboard's ssh login per
# poll) was restarted and the dashboard stopped. Rank 0 binds (it runs the HTTP server). Measured
# 12,500 since 2026-10-22: native EXL3 with FP8 kv_b bottomed at 9.19 GB on rank 0 at 12,000
# (results/glm-native-perf-20261022), so 12,500 (7.1 MB a slot) projects to ~5.6 GB, the hybrid's
# 12,400 margin (4.97 GiB).
GLM_SLOTS=12500
# Routed experts in EXL3 3-bit (bertholomus/GLM-5.3-EXL3-3.0bpw, packed per rank by
# tools/glm_pack_exl3.py): 7.1 MB a slot against MXFP4's 10.0, so the same memory as 8,600 MXFP4
# slots holds 11,756 (61% of the decoder layers' routed experts). Fixed/dense weights and
# the separate 4-bit MTP arena now come from this same EXL3 checkpoint.
GLM_EXPERT_FORMAT=exl3
GLM_EXL3_DIR=/home/ryan/models/glm-exl3-packs
GLM_EXL3_PACK=/packs/glm-exl3-experts-r{rank}of2.bin
# Prefill MoE kernel: smem:WxMTWxNTBxPF (one trellis decode a 64-row block, shared through shared
# memory) is bit-identical to the default tile kernel and 8% faster on the MoE at 4,096-row chunks
# (no faster at 2,048: tools/bench_exl3_prefill.py, 2026-10-20).
# ca: the smem kernel with cp.async-staged trellis words; bit-identical, -2 ms a layer
# (results/glm-prefill-ca-20261023/). smem:4x1x8x2 reverts.
GLM_EXL3_PREFILL=ca:4x1x8x4
GLM_RESERVE_GB=20
GLM_MISS=substitute
# Resident set from tools/glm_trace.py (same path on both nodes); unset = a per-layer placeholder.
# GLM_RESIDENT=/app/results/glm-trace-.../resident-7500.json
# MTP speculative decoding: drafts a round (0 = off).
GLM_MTP=1
GLM_MTP_DEPTH=7
GLM_MTP_DYN=0.9
# Calibrated gate (engine/glm/mtp_cal.py): stop on measured acceptance instead of the MTP's
# own chain probability, threshold at a verify row's break-even. bench.py 2026-10-23, same
# prompts, DB restored a boot: code 20.24 tok/s at 3.42 drafts/2.52 accepted a round against
# the flat 0.9's 19.60 at 3.27/2.38; prose 15.31 against 15.10. One boot an arm, so the code
# gain (+3%) is the size of the noise; the ladder gate measured -10/-12% on the same run.
# Overrides GLM_MTP_DYN when set.
GLM_MTP_CAL=0.36
# Copy rounds: once the last 8 tokens repeat earlier text, verify its continuation instead of MTP
# drafts (lossless). Rewriting a file +9% after drift, tool re-runs +2%, prose none (2026-10-23).
GLM_LOOKUP=8
# Published 6-bit EXL3 head, from this checkpoint.
GLM_HEAD=exl3
# Published FP16 kv_b quantized at load to FP8 (Quark's 128x128 blocks): decode's absorbs 72-83 ->
# 43-45 us a layer, 0.58 GB a rank less (docs/glm53-native-exl3.md). INT8 embedding is converted
# from this checkpoint's own embedding; routers remain FP32.
GLM_KVB_FP8=1
GLM_EMBED_8BIT=int8
# Reply cap when the request sends no max_tokens (Open WebUI doesn't); clamped to the context
# left, so in effect "until the context is full" (DS V4.1's default too). 4096 cut off HTML
# pages with their reasoning. The KV cache is sized by GLM_MAX_SEQ, not by this.
# Expert adaptation: always once after prefill (before the first decode step), then every
# GLM_ADAPT_EVERY generated tokens; swaps capped per adaptation. 512 after prefill: 2,048 cost
# 1.7 s more TTFT on a 3.3K prompt for the same decode miss (36.2% vs 36.3%).
GLM_ADAPT=1
GLM_ADAPT_EVERY=200
GLM_ADAPT_PREFILL_SWAPS=512
# Prefill a new message of 4-512 tokens twice, loading what it routes to in between: Japanese
# turns against every expert exact, EXL3 KL 0.096 -> 0.074 and 0.110 -> 0.084 (a new topic on a
# cached conversation), MXFP4 0.144 -> 0.136 and 0.211 -> 0.154; +1-3 s to the first token.
GLM_PREFILL_TWO_PASS=512
# ... unless the first pass computed less than this share of the new tokens' routed weight on
# substitutes. Measured 2026-10-20 at 12,000 EXL3 slots: fresh topics missed 27-41% in their first
# pass, follow-up turns on a settled conversation 12-14%, and the second pass costs 1-3 s of first
# token; its quality gain was measured at the fresh-topic miss rates, not at 12-14%.
GLM_PREFILL_TWO_PASS_MISS=0.15
GLM_ADAPT_DECODE_SWAPS=256
# Steadier history and a clear margin: fewer, more useful swaps (churn is logged).
GLM_ADAPT_DECAY=0.9
GLM_ADAPT_MARGIN=1.5
GLM_STATS_DB=/app/results/glm-stats/db.pt
# GLM's template knows low / high; anything else (the server's own default, medium) renders
# "Reasoning Effort: Max". Low keeps chat answers short. A request overrides it with
# reasoning.effort, reasoning_effort or chat_template_kwargs.reasoning_effort (first wins).
GLM_REASONING_EFFORT=low
# Background period: every 2 rounds rank 0 plans the victims kept hot reads replace (below). It
# reads nothing of its own (GLM_BG_READS=0): next to kept hot reads, background reads cost speed
# for little (8 reads, 0.2 uncapped, kept: 7.9 tok/s KL 0.152 vs none + cap 10: 9.19 tok/s KL 0.164,
# one-row eval). Prefill prefetch (GLM_PF) is off: on a 3.3K-token prompt it took prefill miss
# 33% -> 25% but TTFT 23.7 -> 32.8 s, decode miss unchanged.
GLM_BG=1
GLM_BG_EVERY=2
GLM_BG_READS=0
GLM_PF=0
GLM_PF_SLOTS=16
# Hot misses: per layer, the heaviest missed pick with router weight share >= 0.2 is read now,
# computed exactly, and kept resident over a cold, lowest-ranked victim (GLM_HOT_KEEP victims
# queued a period); at most GLM_HOT_MAX reads a step, paced evenly over the layers
# (GLM_HOT_PACE=1: by the i-th of n MoE layers, at most ceil(cap * (i+1)/n)). Tokyo eval in
# 3-row (MTP depth 2) steps, KL to exact / top-1, and probe tok/s:
#   target-4 controller, bg 8, not kept   0.299 / 86.4%   10.7
#   0.2, cap 10, first come, kept         0.235 / 88.7%    9.19
#   0.2, cap 10, paced, kept              0.212 / 89.9%    9.29
#   0.2, cap 12, paced, kept              0.189 / 89.2%    9.03
#   0.2, cap 16, paced, kept              0.171 / 90.9%    8.95
#   0.2, no cap, kept                     0.172 / 90.3%    8.24
#   0.15, cap 16, paced, kept             0.194 / 90.1%    -- (1.8x the reads; light misses churn residents)
#   0.25, cap 10, kept                    0.257 (one-row)  10.39
GLM_HOT=1
GLM_HOT_MIN=0.2
GLM_HOT_TARGET=0
GLM_HOT_MAX=16
GLM_HOT_PACE=1
GLM_HOT_KEEP=32

# A cycle of 1-8 tokens about to repeat a 12th time: that token is excluded and the next one
# drawn again (docs/gotchas.md: substituted experts open repetition loops; the model holds them).
GLM_LOOP_BREAK=12

GLM_INDEX_BUCKETS=0
TIDE_ENGINE=rust
